Attack Vector Spotlight

The Reentrancy Threat

Reentrancy occurs when external contract calls are allowed to make new calls to the calling contract before the initial execution is complete. This is the most infamous smart contract vulnerability.

mapping(address => uint) public balances;

function withdraw() public {
    uint bal = balances[msg.sender];
    require(bal > 0);
    // VULNERABLE: State updated AFTER external call
    (bool sent, ) = msg.sender.call{value: bal}("");
    require(sent, "Failed to send Ether");
    balances[msg.sender] = 0;
}

mapping(address => uint) public balances;

function withdraw() public {
    uint bal = balances[msg.sender];
    require(bal > 0);
    // SECURE: Checks, Effects, Interactions pattern
    balances[msg.sender] = 0; // State updated BEFORE external call
    (bool sent, ) = msg.sender.call{value: bal}("");
    require(sent, "Failed to send Ether");
}
